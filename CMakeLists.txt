cmake_minimum_required(VERSION 3.24)
project(TexUtil VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(FetchContent)
set(FASTNOISE2_TOOLS OFF CACHE BOOL "" FORCE)
set(FASTNOISE2_TESTS OFF CACHE BOOL "" FORCE)
set(FASTNOISE2_UTILITY OFF CACHE BOOL "" FORCE)
set(FASTNOISE2_STRICT_FP ON CACHE BOOL "" FORCE)
FetchContent_Declare(FastNoise2 GIT_REPOSITORY https://github.com/Auburn/FastNoise2.git GIT_TAG 903c1f2d2f9d53ddce94cd223f32727d9ab3aeaa)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(FastNoise2 json)
# FastSIMD declares x86 __regcall attributes on ARM. Apple Clang ignores them.
if(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang" AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  target_compile_options(FastSIMD_FastNoise PRIVATE -Wno-ignored-attributes)
endif()
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(SYSTEM_PNG QUIET IMPORTED_TARGET libpng)
endif()
if(TARGET PkgConfig::SYSTEM_PNG)
  set(TEXUTIL_PNG PkgConfig::SYSTEM_PNG)
else()
  find_package(PNG REQUIRED)
  set(TEXUTIL_PNG PNG::PNG)
endif()
find_package(Threads REQUIRED)
add_library(texutil_core src/image.cpp src/graph.cpp src/nodes.cpp src/catalog.cpp src/advanced_catalog.cpp src/advanced.cpp src/erosion.cpp src/presets.cpp src/fields.cpp src/sheet.cpp src/materialx.cpp src/imports.cpp src/material_recipe.cpp src/export_bake.cpp)
target_include_directories(texutil_core PUBLIC include)
target_link_libraries(texutil_core PUBLIC nlohmann_json::nlohmann_json PRIVATE FastNoise ${TEXUTIL_PNG} Threads::Threads)
target_compile_options(texutil_core PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic>)
include(cmake/Models.cmake)
include(cmake/Filament.cmake)
add_executable(texutil src/main.cpp)
target_link_libraries(texutil PRIVATE texutil_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(texutil_export_bake_tests tests/export_bake_tests.cpp)
  target_link_libraries(texutil_export_bake_tests PRIVATE texutil_core)
  add_test(NAME export_bake COMMAND texutil_export_bake_tests ${CMAKE_CURRENT_BINARY_DIR}/export-bake-test)
  add_executable(texutil_model_tests tests/model_tests.cpp)
  target_link_libraries(texutil_model_tests PRIVATE texutil_core)
  add_test(NAME model_operations COMMAND texutil_model_tests ${CMAKE_CURRENT_BINARY_DIR}/model-test ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(texutil_tests tests/tests.cpp)
  target_link_libraries(texutil_tests PRIVATE texutil_core)
  add_test(NAME texutil_tests COMMAND texutil_tests)
  add_executable(texutil_advanced_tests tests/advanced_tests.cpp)
  target_link_libraries(texutil_advanced_tests PRIVATE texutil_core)
  add_test(NAME advanced_nodes COMMAND texutil_advanced_tests)
  add_executable(texutil_sheet_tests tests/sheet_tests.cpp)
  target_link_libraries(texutil_sheet_tests PRIVATE texutil_core)
  add_test(NAME native_sheets COMMAND texutil_sheet_tests)
  add_executable(texutil_materialx_tests tests/materialx_tests.cpp)
  target_link_libraries(texutil_materialx_tests PRIVATE texutil_core)
  add_test(NAME materialx_export COMMAND texutil_materialx_tests ${CMAKE_CURRENT_BINARY_DIR}/materialx-test)
  add_executable(texutil_import_tests tests/import_tests.cpp)
  target_link_libraries(texutil_import_tests PRIVATE texutil_core)
  add_test(NAME graph_imports COMMAND texutil_import_tests ${CMAKE_CURRENT_BINARY_DIR}/import-test)
  foreach(example nodes-gallery presets-gallery erosion effects-gallery forest snow handled-glass)
    add_test(NAME ${example} COMMAND texutil ${CMAKE_CURRENT_SOURCE_DIR}/examples/${example}.json --size 64 --out ${CMAKE_CURRENT_BINARY_DIR}/${example}-test --threads 2 --json)
  endforeach()
  add_test(NAME portable_samples COMMAND ${CMAKE_COMMAND} -DTEXUTIL=$<TARGET_FILE:texutil> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DWORK=${CMAKE_CURRENT_BINARY_DIR}/sample-tests -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/samples.cmake)
  add_test(NAME cli_help COMMAND texutil --help)
  add_test(NAME cli_validate COMMAND texutil validate ${CMAKE_CURRENT_SOURCE_DIR}/examples/stone.json)
  add_test(NAME cli_integration COMMAND ${CMAKE_COMMAND} -DTEXUTIL=$<TARGET_FILE:texutil> -DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-tests -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
  add_test(NAME wood_example COMMAND texutil ${CMAKE_CURRENT_SOURCE_DIR}/examples/wood.json --size 128 --out ${CMAKE_CURRENT_BINARY_DIR}/wood-test --threads 2 --json)
endif()
install(TARGETS texutil RUNTIME DESTINATION bin)
